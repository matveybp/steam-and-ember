```python
from dataclasses import dataclass

@dataclass(frozen=True)
class Coffee:
    name: str
    temperature: int = 67

    def serve(self, with_sugar: bool = False) -> str:
        suffix = " with sugar" if with_sugar else ""
        return f"{self.name} at {self.temperature}°C{suffix}"
```
